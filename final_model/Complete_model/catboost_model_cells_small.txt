def prefix_features(df, prefix):
    df = df.copy()

    feature_cols = [
        c for c in df.columns
        if c != "pair_key"
    ]

    df = df.rename(
        columns={
            c: f"{prefix}_{c}"
            for c in feature_cols
        }
    )

    return df

    if path.suffix.lower() == ".csv":

        return pd.read_parquet(path)
    raise ValueError(f"Unsupported file: {path}")

    matches = [
        f for f in files

    ]

        raise RuntimeError(
